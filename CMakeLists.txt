function(meter_sources output manifest root key)
    string(JSON count LENGTH "${manifest}" "${key}")
    set(sources)
    if(count GREATER 0)
        math(EXPR end "${count} - 1")
        foreach(i RANGE 0 ${end})
            string(JSON source GET "${manifest}" "${key}" ${i})
            get_filename_component(path "${root}/${source}" REALPATH)
            file(RELATIVE_PATH relative "${root}" "${path}")
            if(relative MATCHES "^[.][.](/|$)" OR IS_ABSOLUTE "${source}" OR NOT EXISTS "${path}")
                message(FATAL_ERROR "Source escapes package or is missing: ${source}")
            endif()
            list(APPEND sources "${path}")
        endforeach()
    endif()
    set(${output} "${sources}" PARENT_SCOPE)
endfunction()
function(meter_library target key scope)
    if(scope STREQUAL "product")
        meter_sources(sources "${product_manifest}" "${METER_PRODUCT_ROOT}" "${key}")
    else()
        meter_sources(sources "${platform_manifest}" "${CMAKE_CURRENT_SOURCE_DIR}" "${key}")
    endif()
    add_library(${target} STATIC ${sources})
    target_link_libraries(${target} PUBLIC meter_contracts)
    if(scope STREQUAL "product")
        target_include_directories(${target} PUBLIC "${METER_PRODUCT_ROOT}")
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endfunction()
