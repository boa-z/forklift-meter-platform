file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cmake/sources.json" source_manifest)
function(meter_library target key)
    string(JSON count LENGTH "${source_manifest}" "${key}")
    math(EXPR end "${count} - 1")
    set(sources)
    foreach(i RANGE 0 ${end})
        string(JSON source GET "${source_manifest}" "${key}" ${i})
        list(APPEND sources "${source}")
    endforeach()
    add_library(${target} STATIC ${sources})
    target_link_libraries(${target} PUBLIC meter_contracts)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endfunction()
