function(hmi_sources output manifest root key)
    string(JSON count LENGTH "${manifest}" "${key}")
    set(sources)
    if(count GREATER 0)
        math(EXPR end "${count} - 1")
        foreach(i RANGE 0 ${end})
            string(JSON source GET "${manifest}" "${key}" ${i})
            get_filename_component(path "${root}/${source}" REALPATH)
            file(RELATIVE_PATH relative "${root}" "${path}")
            if(relative MATCHES "^[.][.](/|$)" OR IS_ABSOLUTE "${source}" OR NOT EXISTS "${path}")
                message(FATAL_ERROR "Source escapes package or is missing: ${source}")
            endif()
            list(APPEND sources "${path}")
        endforeach()
    endif()
    set(${output} "${sources}" PARENT_SCOPE)
endfunction()
function(hmi_library target key scope)
    if(scope STREQUAL "product")
        hmi_sources(sources "${product_manifest}" "${HMI_PRODUCT_ROOT}" "${key}")
    else()
        hmi_sources(sources "${platform_manifest}" "${CMAKE_CURRENT_SOURCE_DIR}" "${key}")
    endif()
    add_library(${target} STATIC ${sources})
    target_link_libraries(${target} PUBLIC hmi_contracts)
    if(scope STREQUAL "product")
        target_include_directories(${target} PUBLIC "${HMI_PRODUCT_ROOT}")
        foreach(key include_dirs host_include_dirs)
            string(JSON dirs_count ERROR_VARIABLE dirs_error LENGTH "${product_manifest}" "${key}")
            if(NOT dirs_error AND dirs_count GREATER 0)
                hmi_sources(dirs "${product_manifest}" "${HMI_PRODUCT_ROOT}" "${key}")
                foreach(dir IN LISTS dirs)
                    if(NOT IS_DIRECTORY "${dir}")
                        message(FATAL_ERROR "Product include directory is not a directory: ${dir}")
                    endif()
                endforeach()
                target_include_directories(${target} PUBLIC ${dirs})
            endif()
        endforeach()
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endfunction()
